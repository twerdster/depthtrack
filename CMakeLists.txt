cmake_minimum_required(VERSION 3.20)
project(DepthTrack VERSION 0.1.0 LANGUAGES C CXX)
option(DEPTHTRACK_BUILD_EXAMPLES "Build CSV command-line example" ON)
option(DEPTHTRACK_BUILD_TESTS "Build native tests" ON)
option(DEPTHTRACK_BUILD_SHARED "Build a shared rather than static library" OFF)
set(CMAKE_CXX_EXTENSIONS OFF)
if(DEPTHTRACK_BUILD_SHARED)
  add_library(depthtrack SHARED src/tracker.cpp)
  target_compile_definitions(depthtrack PUBLIC DEPTHTRACK_SHARED PRIVATE DEPTHTRACK_BUILD)
else()
  add_library(depthtrack STATIC src/tracker.cpp)
endif()
add_library(DepthTrack::depthtrack ALIAS depthtrack)
target_compile_features(depthtrack PUBLIC cxx_std_17)
target_include_directories(depthtrack PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
set_target_properties(depthtrack PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(depthtrack PRIVATE /W4 /fp:strict)
else()
  target_compile_options(depthtrack PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
endif()
if(DEPTHTRACK_BUILD_EXAMPLES)
  add_executable(depthtrack_csv examples/tracking_csv.cpp)
  target_link_libraries(depthtrack_csv PRIVATE DepthTrack::depthtrack)
endif()
if(DEPTHTRACK_BUILD_TESTS)
  enable_testing()
  add_executable(depthtrack_native_tests tests/native/test_tracker.cpp)
  target_link_libraries(depthtrack_native_tests PRIVATE DepthTrack::depthtrack)
  add_test(NAME native_unit COMMAND depthtrack_native_tests)
  add_test(NAME native_golden_csv COMMAND depthtrack_native_tests --golden ${CMAKE_CURRENT_SOURCE_DIR}/samples/handmesh/tracking_fixture.csv)
  add_executable(depthtrack_c_header_test tests/native/test_c_api.c)
  target_link_libraries(depthtrack_c_header_test PRIVATE DepthTrack::depthtrack)
  set_target_properties(depthtrack_c_header_test PROPERTIES LINKER_LANGUAGE CXX)
  add_test(NAME c_header COMMAND depthtrack_c_header_test)
  if(DEPTHTRACK_BUILD_EXAMPLES)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
      add_test(NAME native_golden_json COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/native/test_golden.py --cli $<TARGET_FILE:depthtrack_csv> --fixture ${CMAKE_CURRENT_SOURCE_DIR}/samples/handmesh/tracking_fixture.json)
    endif()
  endif()
endif()
include(GNUInstallDirs)
install(TARGETS depthtrack EXPORT DepthTrackTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT DepthTrackTargets NAMESPACE DepthTrack:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/DepthTrack)
