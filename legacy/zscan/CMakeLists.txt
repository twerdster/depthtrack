cmake_minimum_required(VERSION 3.16)
project(depthtrack_zscan LANGUAGES CXX)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc video features2d)
add_executable(depthtrack_zscan run_depth.cpp source-audit/forest_adapter.cpp algorithm-audit/depth_geometry.cpp)
target_compile_features(depthtrack_zscan PRIVATE cxx_std_17)
target_include_directories(depthtrack_zscan PRIVATE ${OpenCV_INCLUDE_DIRS} original source-audit/compat algorithm-audit/compat)
target_link_libraries(depthtrack_zscan PRIVATE ${OpenCV_LIBS})
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(depthtrack_zscan PRIVATE -O2 -ffp-contract=off -fno-fast-math)
endif()
